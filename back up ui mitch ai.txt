<script>
    async function sendMessage() {
        const inputField = document.getElementById("user-input");
        const chatBox = document.getElementById("chat-box");
        const text = inputField.value.trim();
        
        if (!text) return;

        chatBox.innerHTML += `<div class="message user-msg">${text}</div>`;
        inputField.value = "";
        chatBox.scrollTop = chatBox.scrollHeight;

        try {
            const response = await fetch(`http://127.0.0.1:8000/chat?user_input=${encodeURIComponent(text)}`, {
                method: "POST",
                headers: { "Accept": "application/json" }
            });

            const data = await response.json();
            const formattedResponse = data.response.replace(/\n/g, "<br>");
            chatBox.innerHTML += `<div class="message ai-msg">${formattedResponse}</div>`;
        } catch (error) {
            chatBox.innerHTML += `<div class="message ai-msg" style="color: #ff5252;">Error 500: Server unreachable. Probably because your last idea lacked asymmetric upside. (Or you forgot to run the python server).</div>`;
        }
        
        chatBox.scrollTop = chatBox.scrollHeight;
    }

    async function clearMemory() {
        try {
            await fetch('http://127.0.0.1:8000/clear_memory', { method: 'POST' });
            const chatBox = document.getElementById("chat-box");
            chatBox.innerHTML = '<div class="message ai-msg">Memory wiped. We start fresh. Do not disappoint me this time.</div>';
        } catch (error) {
            console.error("Failed to clear memory", error);
        }
    }

    function handleEnter(event) {
        if (event.key === "Enter") {
            sendMessage();
        }
    }
</script>
